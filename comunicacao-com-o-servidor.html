// ex22/Ex22Activity.kt - uma primeira requisição HTTP ao servidor
// -------------------------------------------------------------------------
// O servidor JSON do RdvMedecins deve estar em execução (porta 8080). Pedimos a ele
// um captcha: GET /api/captcha -> { "id": "...", "svg": "<svg ...>" }.
//
//   - OkHttp, o cliente HTTP mais usado no Android: descrevemos a
//     requisição (Request), o cliente a executa (newCall(...).execute()) e
//     devolve a resposta: seu código de status (code), seus cabeçalhos
//     (headers), seu corpo (body);
//   - execute() BLOQUEIA até a resposta: o Android proíbe chamá-lo
//     na thread da tela. withContext(Dispatchers.IO)
//     o executa em outra thread, e a corrotina ESPERA o resultado;
//   - Json.parseToJsonElement(text): o corpo JSON -> objetos Kotlin;
//   - o endereço do servidor depende de onde a aplicação executa
//     (emulador, celular...): cf. common/Config.kt;
//   - três estados a exibir: carregando, sucesso, erro;
//   - a imagem do captcha é SVG: cf. common/SvgImage.kt.
// =========================================================================

// o que GET /api/captcha devolve
data class Captcha(val id: String, val svg: String)

// um cliente HTTP: um objeto caro de criar, que é criado UMA ÚNICA vez
val client = OkHttpClient()

// a requisição: uma função SUSPEND, que devolve o captcha ou lança uma exceção
suspend fun readCaptcha(): Captcha = withContext(Dispatchers.IO) {
    val request = Request.Builder().url("${Config.serverUrl}/api/captcha").build()
    client.newCall(request).execute().use { response ->
        // OkHttp NÃO lança exceção para uma resposta de erro (4xx,
        // 5xx): o código de status é testado
        if (!response.isSuccessful) {
            throw Exception("the server responded ${response.code} ${response.message}")
        }
        val json = Json.parseToJsonElement(response.body!!.string()).jsonObject
        Captcha(json["id"]!!.jsonPrimitive.content, json["svg"]!!.jsonPrimitive.content)
    }
}

@Composable
fun CaptchaPage() {
    var captcha by remember { mutableStateOf<Captcha?>(null) }
    var loading by remember { mutableStateOf(true) }
    var error by remember { mutableStateOf<String?>(null) }
    var requestCount by remember { mutableIntStateOf(0) }
    // o número do PEDIDO: alterá-lo relança o efeito abaixo
    var demand by remember { mutableIntStateOf(0) }

    // a requisição, assim que a tela aparece (pedido = 0), depois a cada
    // novo pedido. Se um pedido chega antes da resposta ao
    // anterior, a corrotina anterior é CANCELADA: sua resposta
    // obsoleta nunca será exibida
    LaunchedEffect(demand) {
        loading = true
        error = null
        try {
            captcha = readCaptcha()
            requestCount++
        } catch (e: CancellationException) {
            throw e // a corrotina foi cancelada (novo pedido): nada a exibir
        } catch (e: Exception) {
            // nenhuma resposta (servidor parado, endereço errado...), ou erro
            error = e.message ?: e.toString()
            captcha = null
        }
        loading = false
    }

    ExamplePage("Example 22 - a first HTTP request") {
        Text("Server: ${Config.serverUrl}", color = Color.Gray)
        Spacer(Modifier.height(12.dp))
        if (loading) LinearProgressIndicator(Modifier.fillMaxWidth())
        error?.let {
            Card(colors = CardDefaults.cardColors(containerColor = Color(0xFFFFEBEE))) {
                Text("Unable to get a captcha: $it.\nIs the RdvMedecins JSON server running (port 8080)?", Modifier.padding(12.dp))
            }
        }
        val c = captcha
        if (c != null && !loading) {
            SvgImage(
                c.svg,
                height = 60.dp,
                modifier = Modifier.border(1.dp, Color(0xFFE0E0E0), RoundedCornerShape(6.dp)).padding(4.dp),
            )
            Spacer(Modifier.height(8.dp))
            Text("id: ${c.id}", fontFamily = FontFamily.Monospace, fontSize = 11.sp)
        }
        Spacer(Modifier.height(12.dp))
        Button(onClick = { demand++ }, enabled = !loading) { Text("Another image") }
        Spacer(Modifier.height(8.dp))
        Text("$requestCount successful request(s)", color = Color.Gray)
    }
}
    • Linha 66: um cliente HTTP, objeto caro de criar, que é criado UMA ÚNICA VEZ;
    • Linhas 69–71: a requisição, uma função SUSPEND. A requisição ([Request]) é descrita; o cliente a executa ([newCall(...).execute()]) e retorna a resposta: seu código de status, cabeçalhos e corpo. [execute()] BLOQUEIA até a resposta: o [Android] proíbe chamá-lo na thread de execução da tela ([NetworkOnMainThreadException]). [withContext(Dispatchers.IO)] o executa em uma thread reservada para E/S, e a corrotina ESPERA pelo resultado;
    • Linhas 74–75: o [OkHttp] não lança uma exceção para uma resposta de erro (4xx, 5xx); em vez disso, é preciso verificar o código de status. No entanto, ele lança uma [IOException] se não houver nenhuma resposta (servidor fora do ar, endereço incorreto);
    • linhas 77–78: o corpo JSON, lido “manualmente”: [jsonObject], [jsonPrimitive.content];
    • linhas 103–116: a requisição é disparada por um [LaunchedEffect] cuja chave é um número de REQUISIÇÃO: o botão “Outra imagem” o incrementa, o que reinicia o efeito. Se uma requisição chegar antes de a anterior ter respondido, a corrotina anterior é CANCELADA: sua resposta, agora obsoleta, nunca será exibida;
    • linhas 122–136: três estados a exibir: em andamento (uma barra de progresso), erro e sucesso.

// -------------------------------------------------------------------------
// A imagem do captcha é feita pelo servidor, em formato SVG (um texto
// que descreve um desenho). O Android não consegue exibi-la sozinho: a
// biblioteca AndroidSVG a desenha em uma imagem (um Bitmap), que

package fr.stahe.examples.common

import android.graphics.Bitmap
import android.graphics.Canvas
import androidx.compose.foundation.Image
import androidx.compose.foundation.layout.height
import androidx.compose.runtime.Composable
import androidx.compose.runtime.remember
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.asImageBitmap
import androidx.compose.ui.layout.ContentScale
import androidx.compose.ui.unit.Dp
import androidx.compose.ui.unit.dp
import com.caverock.androidsvg.SVG

@Composable
fun SvgImage(svg: String, modifier: Modifier = Modifier, height: Dp = 50.dp, description: String? = null) {
    // o desenho só é refeito se o texto SVG mudar (remember(svg))
    val image = remember(svg) {
        val document = SVG.getFromString(svg)
        // desenhado 3 vezes maior que seu tamanho "natural": uma imagem nítida
        val scale = 3f
        val width = (document.documentWidth.takeIf { it > 0 } ?: 150f)
        val tall = (document.documentHeight.takeIf { it > 0 } ?: 50f)
        val bitmap = Bitmap.createBitmap((width * scale).toInt(), (tall * scale).toInt(), Bitmap.Config.ARGB_8888)
        val canvas = Canvas(bitmap)
        canvas.scale(scale, scale)
        document.renderToCanvas(canvas)
        bitmap.asImageBitmap()
    }
    Image(image, contentDescription = description, contentScale = ContentScale.Fit, modifier = modifier.height(height))
}

    // =========================================================================
// common/HttpClient.kt - um cliente HTTP que gerencia COOKIES
// -------------------------------------------------------------------------
// O servidor RdvMedecins coloca o token JWT em um COOKIE (httpOnly).
// Em um navegador, é o navegador que armazena os cookies e os devolve
// a cada requisição: o código da aplicação não tem nada a fazer.
// Em um celular, NÃO há navegador: cabe à aplicação
// fazer o papel dele, ou seja:
//   1. ler o cabeçalho "Set-Cookie" de cada resposta e armazenar os cookies;
//   2. devolvê-los no cabeçalho "Cookie" de cada requisição.
// A biblioteca OkHttp faz isso por nós, se lhe dermos um CookieJar
// (um "pote de cookies"): uma classe com duas funções, saveFromResponse
// (armazenar) e loadForRequest (retirar). Esta também armazena os cookies
// no dispositivo (SharedPreferences): continuamos logados quando paramos
// e reiniciamos a aplicação.
// =========================================================================
package fr.stahe.examples.common

import android.content.Context
import android.content.SharedPreferences
import androidx.core.content.edit
import okhttp3.Cookie
import okhttp3.CookieJar
import okhttp3.HttpUrl
import okhttp3.HttpUrl.Companion.toHttpUrl
import okhttp3.OkHttpClient
import java.util.concurrent.TimeUnit

class DeviceCookieJar(private val storage: SharedPreferences) : CookieJar {
    // nome do cookie -> o cookie (seu valor, seu domínio, sua expiração...)
    private val cookies = mutableMapOf<String, Cookie>()

    init {
        // os cookies armazenados durante um uso anterior: cada um
        // é armazenado na forma de seu cabeçalho Set-Cookie
        val base = Config.serverUrl.toHttpUrl()
        for (text in storage.getStringSet(KEY, emptySet()).orEmpty()) {
            Cookie.parse(base, text)?.let { cookies[it.name] = it }
        }
    }

    // a resposta traz cookies: nós os armazenamos. Um cookie já expirado
    // (logout: o servidor o apaga dessa forma) é removido
    @Synchronized
    override fun saveFromResponse(url: HttpUrl, cookies: List<Cookie>) {
        for (c in cookies) {
            if (c.expiresAt < System.currentTimeMillis() || c.value.isEmpty()) {
                this.cookies.remove(c.name)
            } else {
                this.cookies[c.name] = c
            }
        }
        storage.edit { putStringSet(KEY, this@DeviceCookieJar.cookies.values.map { it.toString() }.toSet()) }
    }

    // uma requisição sai: os cookies que a ela dizem respeito (endereço, caminho), não expirados
    @Synchronized
    override fun loadForRequest(url: HttpUrl): List<Cookie> =
        cookies.values.filter { it.expiresAt >= System.currentTimeMillis() && it.matches(url) }

    companion object {
        private const val KEY = "rdvmedecins.cookies"
    }
}

// o cliente HTTP dos exemplos: um único para toda a aplicação
fun createHttpClient(context: Context): OkHttpClient = OkHttpClient.Builder()
    .cookieJar(DeviceCookieJar(context.getSharedPreferences("examples", Context.MODE_PRIVATE)))
    // nenhuma resposta após 10 segundos: a requisição falha
    .connectTimeout(10, TimeUnit.SECONDS)
    .readTimeout(10, TimeUnit.SECONDS)
    .build()
    • linha 29: um [CookieJar] tem duas funções: [saveFromResponse] (armazenar os cookies de uma resposta) e [loadForRequest] (recuperar os de uma requisição);
    • linhas 33–38: na inicialização, os cookies armazenados durante uma sessão anterior — salvos no dispositivo ([SharedPreferences], um pequeno arquivo de chave-valor; veja o Exemplo 20) — são recuperados na forma de seu cabeçalho [Set-Cookie];
    • linhas 44–53: um cookie que já expirou ou está vazio é removido; é assim que o servidor exclui o cookie no logout; [@Synchronized]: o [OkHttp] pode chamar essas funções a partir de várias threads ao mesmo tempo;
    • linhas 67–72: o cliente HTTP dos exemplos, com seu jar e seus timeouts.

// Uma ação assíncrona (fazer login, salvar...) tem um estado que
// a tela deve mostrar: está EM ANDAMENTO (botão desabilitado)? terminou

//     val loginCommand = remember { Command() }
//     scope.launch { loginCommand.execute { logIn() } }    // lança a ação
//     loginCommand.inProgress, loginCommand.error          // seu estado
//
// Suas duas propriedades são estados observados pelo Compose: a tela é
// redesenhada quando a ação começa e quando termina.

package fr.stahe.examples.ex23

import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue
import kotlinx.coroutines.CancellationException

class Command {
    var inProgress by mutableStateOf(false)
        private set
    var error by mutableStateOf<String?>(null)
        private set

    // executa a ação, se outra não estiver já em andamento
    suspend fun execute(action: suspend () -> Unit) {
        if (inProgress) return // sem envio duplo (pressionar o botão duas vezes)
        inProgress = true
        error = null
        try {
            action()
        } catch (e: CancellationException) {
            throw e // uma corotina cancelada deve permanecer cancelada
        } catch (e: Exception) {
            error = e.message ?: e.toString() // o erro é MANTIDO, para ser exibido
        } finally {
            // finally: executado em TODOS os casos, sucesso ou erro
            inProgress = false
        }
    }

    fun clearError() {
        error = null
    }
}
    • linha 31: impede o envio duplo (toque duplo no botão);
    • linhas 40–42: [finally]: executado em TODOS os casos, sucesso e erro.

// common/ApiError.kt - UMA classe para todos os erros da API
// -------------------------------------------------------------------------
// Todas as respostas de erro do servidor RdvMedecins têm a mesma forma:
//   { "statusCode": 409, "key": "ERRORS.LOGIN_TAKEN", "params": {...}, "fields": {...} }
// Mas uma requisição também pode falhar DE OUTRO MODO: nenhuma resposta
// (rede fora do ar, servidor parado, endereço errado), ou uma resposta que não
// vem do nosso servidor (uma página de erro de um proxy). Convertemos
// tudo isso em UMA classe de exceção, ApiError, que carrega a chave de tradução
// enviada pelo servidor (e, se houver, seus parâmetros e
// o erro de cada campo).
// =========================================================================
package fr.stahe.examples.common

import kotlinx.serialization.json.Json
import kotlinx.serialization.json.JsonObject
import kotlinx.serialization.json.JsonPrimitive
import kotlinx.serialization.json.jsonObject

class ApiError(
    val statusCode: Int, // 0: o servidor não respondeu
    val key: String, // ex.: "ERRORS.LOGIN_TAKEN"
    val params: Map<String, String> = emptyMap(),
    val fields: Map<String, String> = emptyMap(),
) : Exception("$statusCode $key") {

    override fun toString() = "$statusCode $key"

    // companion object: o que pertence à CLASSE (o "static" do Java)
    companion object {
        // uma resposta de erro (4xx, 5xx) -> um ApiError
        fun fromResponse(statusCode: Int, body: String): ApiError {
            // runCatching: o resultado, ou null se o corpo não for JSON
            // (então não foi o nosso servidor que respondeu)
            val json = runCatching { Json.parseToJsonElement(body).jsonObject }.getOrNull()
            val key = (json?.get("key") as? JsonPrimitive)?.content
            if (json == null || key == null) {
                return ApiError(statusCode, if (statusCode >= 500) "ERRORS.SERVER_UNREACHABLE" else "ERRORS.SERVER_ERROR")
            }
            return ApiError(statusCode, key, strings(json["params"]), strings(json["fields"]))
        }

        // um objeto JSON { "name": "ERRORS..." } -> Map<String, String> (vazio caso contrário)
        private fun strings(json: Any?): Map<String, String> =
            (json as? JsonObject)?.mapValues { (_, v) -> (v as? JsonPrimitive)?.content ?: v.toString() } ?: emptyMap()
    }
}

class Api(private val client: OkHttpClient) {

    suspend fun request(method: String, url: String, body: JsonElement? = null): JsonElement? =
        withContext(Dispatchers.IO) {
            // um corpo JSON (o servidor rejeita um POST que não tenha um)
            val httpBody = body?.toString()?.toRequestBody("application/json".toMediaType())
            val request = Request.Builder().url("${Config.serverUrl}/api$url").method(method, httpBody).build()
            try {
                // use: a resposta é fechada (liberada) ao final do bloco
                client.newCall(request).execute().use { response ->
                    val text = response.body?.string().orEmpty()
                    if (!response.isSuccessful) throw ApiError.fromResponse(response.code, text)
                    if (text.isEmpty()) null else Json.parseToJsonElement(text)
                }
            } catch (e: IOException) {
                // NENHUMA resposta (rede, servidor parado, endereço errado)
                throw ApiError(0, "ERRORS.SERVER_UNREACHABLE")
            }
        }

    suspend fun get(url: String) = request("GET", url)

    // um POST sempre tem corpo JSON: {} por padrão
    suspend fun post(url: String, body: JsonElement = JsonObject(emptyMap())) = request("POST", url, body)

    suspend fun put(url: String, body: JsonElement) = request("PUT", url, body)

    suspend fun delete(url: String, body: JsonElement? = null) = request("DELETE", url, body)
}

class RdvMedecins(private val api: Api) {
    suspend fun getDoctors(): List<Person> = api.get("/doctors").read()

    suspend fun getClients(): List<Person> = api.get("/clients").read()

    // Uri.Builder constrói "doctorId=1&day=2030-01-15", codificando o que for necessário
    suspend fun getSchedule(doctorId: Int, day: String): Schedule {
        val params = Uri.Builder()
            .appendQueryParameter("doctorId", "$doctorId")
            .appendQueryParameter("day", day)
            .build().encodedQuery
        return api.get("/schedule?$params").read()
    }

    // clientId: opcional; a chave só é colocada no JSON se não for null
    suspend fun book(day: String, slotId: Int, clientId: Int? = null) {
        api.post(
            "/appointments",
            buildJsonObject {
                put("day", day)
                put("slotId", slotId)
                if (clientId != null) put("clientId", clientId)
            },
        )
    }

    suspend fun cancelAppointment(appointmentId: Int) {
        api.delete("/appointments/$appointmentId")
    }
}

// ex24/ScheduleModel.kt - o "view model" da tela
// -------------------------------------------------------------------------
// O guia de arquitetura do Android recomenda separar:
//   - a VIEW: as funções @Composable, que exibem e repassam as ações
//     do usuário;
//   - o VIEW MODEL (ViewModel): o estado da tela (o médico escolhido,
//     o dia, a agenda recebida...), as ações (escolher, marcar, cancelar),
//     e as chamadas à camada de acesso à API.
// A view deixa de ter lógica: é fácil de ler; o view model
// não tem componentes: é fácil de testar. E ele sobrevive à
// recriação da activity (rotação da tela).
// As ações lançam suas corrotinas em viewModelScope: elas são
// canceladas se o ViewModel for destruído (activity fechada de vez).
//
// Ele também faz o que não queremos reescrever para cada tela:
//   - um CACHE: a agenda de cada (médico, dia) já lida é mantida; voltar
//     a ela a exibe imediatamente, enquanto é lida de novo em segundo plano;
//   - as respostas OBSOLETAS são ignoradas: uma nova leitura CANCELA a
//     anterior (cf. exemplo 19).

// uma mensagem exibida ao usuário
data class Message(val success: Boolean, val text: String)

class ScheduleModel(private val rdvmedecins: RdvMedecins) : ViewModel() {
    // ---- o estado: states do Compose, modificáveis somente aqui ----
    var user by mutableStateOf<User?>(null)
        private set
    var doctors by mutableStateOf(listOf<Person>())
        private set
    var clients by mutableStateOf(listOf<Person>())
        private set
    var doctorId by mutableStateOf<Int?>(null)
        private set
    var day by mutableStateOf(LocalDate.now().plusDays(1).toString()) // amanhã, "2026-10-06"
        private set
    var clientId by mutableStateOf<Int?>(null)
        private set
    var schedule by mutableStateOf<Schedule?>(null)
        private set
    var loading by mutableStateOf(false)
        private set
    var message by mutableStateOf<Message?>(null)
        private set
    private val cache = mutableMapOf<String, Schedule>()
    private var reading: Job? = null

    // ---- valores calculados ----
    val loggedIn get() = user != null
    val isManager get() = user.let { it != null && it.role != Role.USER }

    // ---- as ações ----
    fun changeUser(u: User?) {
        user = u
        doctors = emptyList()
        clients = emptyList()
        schedule = null
        doctorId = null
        cache.clear() // os direitos mudaram: tudo é esquecido
        if (u == null) return
        viewModelScope.launch {
            try {
                doctors = rdvmedecins.getDoctors()
                if (isManager) clients = rdvmedecins.getClients() // um USER receberia 403
            } catch (e: ApiError) {
                message = Message(false, e.toString())
            }
        }
    }

    fun chooseDoctor(id: Int?) {
        doctorId = id
        message = null
        readSchedule()
    }

    fun chooseDay(d: LocalDate) {
        day = d.toString()
        message = null
        readSchedule()
    }

    fun chooseClient(id: Int?) {
        clientId = id
    }

    fun book(s: ScheduleSlot) = act("Slot ${s.timeRange} booked.") {
        rdvmedecins.book(day, s.id, if (isManager) clientId else null)
    }

    fun cancel(s: ScheduleSlot) = act("Appointment at ${s.timeRange} cancelled.") { rdvmedecins.cancelAppointment(s.appointmentId!!) }

    // uma ação, e SEMPRE uma nova leitura da agenda: o SERVIDOR é a referência
    private fun act(success: String, action: suspend () -> Unit) {
        viewModelScope.launch {
            message = try {
                action()
                Message(true, success)
            } catch (e: ApiError) {
                Message(false, e.toString())
            }
            readSchedule()
        }
    }

    private fun readSchedule() {
        val id = doctorId ?: return
        val key = "$id/$day"
        // uma leitura mais antiga ainda está em andamento: ela é cancelada
        reading?.cancel()
        // primeiro o cache (ou nada), depois a leitura
        schedule = cache[key]
        loading = true
        reading = viewModelScope.launch {
            try {
                val read = rdvmedecins.getSchedule(id, day)
                cache[key] = read
                schedule = read
            } catch (e: ApiError) {
                schedule = null
                message = Message(false, e.toString())
            }
            loading = false
        }
    }
}

@Composable
fun LoginBar(api: Api, onUserChange: (User?) -> Unit) {
    var user by remember { mutableStateOf<User?>(null) }
    var captcha by remember { mutableStateOf<Captcha?>(null) }
    var login by rememberSaveable { mutableStateOf("admin") }
    var password by rememberSaveable { mutableStateOf("") }
    var captchaAnswer by rememberSaveable { mutableStateOf("") }
    var error by remember { mutableStateOf<String?>(null) }
    val scope = rememberCoroutineScope()

    fun changeUser(u: User?) {
        user = u
        onUserChange(u)
    }

    suspend fun newCaptcha() {
        captcha = api.get("/captcha").read<Captcha>()
        captchaAnswer = ""
    }

    // na inicialização: já estamos logados?
    LaunchedEffect(Unit) {
        try {
            changeUser(api.get("/auth/me").read<User>())
        } catch (e: ApiError) {
            changeUser(null) // 401: ninguém está logado
            try {
                newCaptcha()
            } catch (e: ApiError) {
                error = "Is the JSON server running (${Config.serverUrl})?"
            }
        }
    }

    suspend fun logIn() {
        error = null
        try {
            val body = buildJsonObject {
                put("login", login.trim())
                put("password", password)
                put("captchaId", captcha?.id)
                put("captcha", captchaAnswer)
            }
            val u = api.post("/auth/login", body).read<User>()
            password = ""
            changeUser(u)
        } catch (e: ApiError) {
            error = e.toString() // "401 LOGIN.ERROR"
            runCatching { newCaptcha() }
        }
    }

    suspend fun logOut() {
        runCatching { api.post("/auth/logout") }
        changeUser(null)
        runCatching { newCaptcha() }
    }

    Card(colors = CardDefaults.cardColors(containerColor = Color(0xFFECEFF1))) {
        val u = user
        if (u != null) {
            Row(Modifier.padding(8.dp), verticalAlignment = Alignment.CenterVertically) {
                Icon(Icons.Default.Person, contentDescription = null)
                Spacer(Modifier.width(8.dp))
                Text("${u.name} (${u.role})", Modifier.weight(1f))
                TextButton(onClick = { scope.launch { logOut() } }) {
                    Icon(Icons.AutoMirrored.Filled.ExitToApp, contentDescription = null)
                    Text("Log out")
                }
            }
        } else {
            Column(Modifier.padding(8.dp)) {
                Row {
                    OutlinedTextField(login, { login = it }, label = { Text("login") }, singleLine = true, modifier = Modifier.weight(1f))
                    Spacer(Modifier.width(8.dp))
                    OutlinedTextField(
                        password,
                        { password = it },
                        label = { Text("password") },
                        singleLine = true,
                        visualTransformation = PasswordVisualTransformation(),
                        modifier = Modifier.weight(1f),
                    )
                }
                Row(verticalAlignment = Alignment.CenterVertically) {
                    // a imagem SVG do captcha, feita pelo NOSSO servidor (cf. exemplo 22)
                    captcha?.let { SvgImage(it.svg, height = 40.dp) }
                    Spacer(Modifier.width(8.dp))
                    OutlinedTextField(captchaAnswer, { captchaAnswer = it }, label = { Text("captcha") }, singleLine = true, modifier = Modifier.weight(1f))
                    IconButton(onClick = { scope.launch { logIn() } }) {
                        Icon(Icons.Default.Check, contentDescription = "Log in")
                    }
                }
                error?.let { Text(it, color = Color.Red) }
            }
        }
    }
}

                    choices = listOf<Int?>(null) + model.clients.map { it.id },
                    text = { id -> model.clients.find { it.id == id }?.fullName ?: "-- patient to book for --" },
                    onChange = { model.chooseClient(it) },
                    modifier = Modifier.fillMaxWidth(),
                )
            }
            model.message?.let { m ->
                Card(colors = CardDefaults.cardColors(containerColor = if (m.success) Color(0xFFE8F5E9) else Color(0xFFFFEBEE))) {
                    Text(m.text, Modifier.padding(10.dp))
                }
            }
            if (model.loading) LinearProgressIndicator(Modifier.fillMaxWidth())
            model.schedule?.let { schedule ->
                // info: a chave enviada pelo servidor quando um paciente já tem uma consulta futura
                schedule.info?.let { Text(it, color = Color(0xFF1976D2)) }
                Text(
                    "Schedule of ${schedule.day} - ${schedule.doctor.fullName}",
                    style = MaterialTheme.typography.titleSmall,
                    modifier = Modifier.padding(vertical = 8.dp),
                )
                for (s in schedule.slots) {
                    SlotRow(s, onBook = { model.book(s) }, onCancel = { model.cancel(s) })
                }

        }

    if (calendarOpen) {

                    calendarOpen = false
                }) { Text("OK") }
            },
        ) { DatePicker(state = state) }
    }

import kotlinx.serialization.json.JsonPrimitive

class Dictionary(context: Context, file: String) {
    // o conteúdo do arquivo JSON, lido uma vez por todas
    private val root: JsonObject = context.assets.open(file).bufferedReader().use { reader ->
        Json.parseToJsonElement(reader.readText()) as JsonObject
    }

    // "FORM.ADDED" -> root["FORM"]["ADDED"]; null se ausente
    private fun find(key: String): String? {
        var node: JsonElement? = root
        for (part in key.split('.')) {
            node = (node as? JsonObject)?.get(part)
        }
        return (node as? JsonPrimitive)?.takeIf { it.isString }?.content
    }

    fun t(key: String, params: Map<String, String> = emptyMap()): String {
        val text = find(key) ?: return key
        // {{name}} -> o valor do parâmetro name (ou {{name}} se não for fornecido)
        return Regex("\\{\\{(\\w+)\\}\\}").replace(text) { m -> params[m.groupValues[1]] ?: m.value }
    }
}

// o dicionário do idioma atual: assets/i18n/<prefix>_<language>.json
@Composable
fun rememberDictionary(prefix: String): Dictionary {
    val context = LocalContext.current
    val language = LocalConfiguration.current.locales[0].language.let { if (it == "fr") "fr" else "en" }
    // lido de novo somente se o idioma mudar
    return remember(language) { Dictionary(context, "i18n/${prefix}_$language.json") }
}

data class KeyMessage(val key: String, val params: Map<String, String> = emptyMap())

class ClientsModel(private val api: Api) : ViewModel() {
    var user by mutableStateOf<User?>(null)
        private set
    var clients by mutableStateOf(listOf<Person>())
        private set

    // o formulário: editedId == null -> criação, caso contrário atualização
    var editedId by mutableStateOf<Int?>(null)
        private set
    private var version = 0

    // os valores digitados, por nome de campo (os nomes esperados pelo servidor)
    var input by mutableStateOf(empty)
        private set

    // o erro de cada campo (uma CHAVE de tradução), devolvido pelo servidor
    var errors by mutableStateOf(mapOf<String, String>())
        private set

    // a mensagem geral do último erro, e a de um sucesso
    var pageError by mutableStateOf<ApiError?>(null)
        private set
    var success by mutableStateOf<KeyMessage?>(null)
        private set

    // envio ao servidor: um Command (exemplo 23), que diz se está em andamento
    val sending = Command()

    val isAdmin get() = user?.role == Role.ADMIN

    suspend fun onUserChange(u: User?) {
        user = u
        clients = emptyList()
        if (u?.role == Role.ADMIN) reload()
    }

    private suspend fun reload() {
        clients = try {
            api.get("/clients").read()
        } catch (e: ApiError) {
            emptyList()
        }
    }

    private fun clearMessages() {
        errors = emptyMap()
        pageError = null
        success = null
    }

    // um campo modificado: seu valor muda, seu erro desaparece
    fun change(field: String, value: String) {
        input = input + (field to value)
        errors = errors - field
    }

    fun newForm() {
        editedId = null
        version = 0
        input = empty
        clearMessages()
    }

    fun edit(c: Person) {
        editedId = c.id
        // a VERSÃO lida com o registro volta com a atualização: se
        // alguém modificou o registro nesse meio tempo, o servidor responderá 409
        // CONCURRENT_UPDATE
        version = c.version
        input = mapOf("title" to c.title, "firstName" to c.firstName, "lastName" to c.lastName, "login" to (c.login ?: ""), "password" to "")
        clearMessages()
    }

    suspend fun save() {
        clearMessages()
        val body = buildJsonObject {
            for ((field, value) in input) put(field, value)
            put("version", version)
        }
        try {
            val id = editedId
            val client: Person = (if (id == null) api.post("/clients", body) else api.put("/clients/$id", body)).read()
            val key = if (id == null) "FORM.ADDED" else "FORM.UPDATED"
            newForm() // formulário esvaziado, mensagens apagadas...
            // ... exceto esta
            success = KeyMessage(key, mapOf("name" to "${client.firstName} ${client.lastName}", "login" to (client.login ?: "")))
            reload()
        } catch (e: ApiError) {
            pageError = e // a mensagem geral
            errors = e.fields // o erro de cada campo
        }
    }

    suspend fun remove(c: Person) {
        clearMessages()
        try {
            api.delete("/clients/${c.id}")
            success = KeyMessage("FORM.DELETED", mapOf("name" to "${c.firstName} ${c.lastName}"))
        } catch (e: ApiError) {
            pageError = e
        }
        reload()
    }

    companion object {
        val empty = mapOf("title" to "Mr", "firstName" to "", "lastName" to "", "login" to "", "password" to "")
    }
}
