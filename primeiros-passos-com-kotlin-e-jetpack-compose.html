    O projeto de exemplo foi criado pelo [Android Studio] (template [Empty Activity], veja o capítulo anterior) e depois completado. Ele é descrito por alguns arquivos escritos em [Kotlin] (os arquivos [.kts], “Kotlin script”).

// ex01/Ex01Activity.kt - um primeiro aplicativo
// -------------------------------------------------------------------------
// Uma tela de um aplicativo Android é uma ACTIVITY: uma classe que
// herda de ComponentActivity, que o sistema cria quando o
// aplicativo é iniciado. Com o Jetpack Compose, a activity quase não faz nada:
// ela entrega sua tela a funções @Composable (setContent).

@Composable
fun CalculatorPage() {
    // o estado: o que o usuário digitou ou escolheu
    var label by rememberSaveable { mutableStateOf("Pen") }
    var priceText by rememberSaveable { mutableStateOf("2.5") } // o texto do campo, como digitado
    var quantity by rememberSaveable { mutableIntStateOf(1) }
    var vatRate by rememberSaveable { mutableDoubleStateOf(20.0) }
    // o histórico das últimas 5 alterações do total com IVA: uma lista
    // NÃO MODIFICÁVEL, substituída por uma nova lista a cada alteração
    var history by rememberSaveable { mutableStateOf(listOf<String>()) }

    // valores CALCULADOS: simples variáveis locais, recalculadas a
    // cada execução da função
    val priceExclVat = price(priceText)
    val totalExclVat = quantity * priceExclVat
    val vatAmount = totalExclVat * vatRate / 100
    val inclVat = totalExclVat + vatAmount

    // TODAS as alterações do estado passam por esta função local: ela
    // anota o total com IVA antes da alteração, aplica a alteração e
    // completa o histórico se o total mudou
    fun change(update: () -> Unit) {
        val old = inclVat
        update()
        // as variáveis de estado JÁ têm seu novo valor
        val newTotal = totalInclVat(price(priceText), quantity, vatRate)
        if (newTotal != old) {
            history = (listOf("${euros.format(old)} -> ${euros.format(newTotal)}") + history).take(5)
        }
    }

    // o título do aplicativo ACOMPANHA o estado, como todo o resto
    TaskTitle("Incl. VAT: ${euros.format(inclVat)}")
    val totalStyle = MaterialTheme.typography.titleMedium.copy(fontWeight = FontWeight.Bold)

    ExamplePage("Example 02 - state") {
        Card {
            Column(Modifier.padding(16.dp)) {
                // um campo CONTROLADO: ele exibe value e chama onValueChange
                // a cada tecla digitada, com o novo texto
                OutlinedTextField(
                    value = label,
                    // trocar o item redefine a quantidade para 1: isso é feito onde
                    // sabemos POR QUE o estado muda
                    onValueChange = { newValue ->
                        change {
                            label = newValue
                            quantity = 1
                        }
                    },
                    label = { Text("Item") },
                    modifier = Modifier.fillMaxWidth(),
                )
                OutlinedTextField(
                    value = priceText,
                    onValueChange = { text -> change { priceText = text } },
                    label = { Text("Unit price excl. VAT") },
                    // o teclado numérico do celular
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Decimal),
                    modifier = Modifier.fillMaxWidth(),
                )
                Spacer(Modifier.height(12.dp))
                Row(verticalAlignment = Alignment.CenterVertically) {
                    // weight(1f): o texto ocupa toda a largura restante
                    Text("Quantity (back to 1 if the item changes)", Modifier.weight(1f))
                    // enabled = false: o botão está desabilitado
                    IconButton(onClick = { change { quantity-- } }, enabled = quantity > 0) {
                        Icon(Icons.Default.KeyboardArrowDown, contentDescription = "minus")
                    }
                    Text("$quantity", style = totalStyle)
                    IconButton(onClick = { change { quantity++ } }) {
                        Icon(Icons.Default.Add, contentDescription = "plus")
                    }
                }
                Spacer(Modifier.height(8.dp))
                // botões dos quais apenas UM é selecionado: a alíquota de IVA
                SingleChoiceSegmentedButtonRow(Modifier.fillMaxWidth()) {
                    rates.forEachIndexed { i, t ->
                        SegmentedButton(
                            selected = t == vatRate,
                            onClick = { change { vatRate = t } },
                            shape = SegmentedButtonDefaults.itemShape(index = i, count = rates.size),
                        ) {
                            Text("${t.toString().removeSuffix(".0")} %")
                        }
                    }
                }
                Spacer(Modifier.height(12.dp))
                OutlinedButton(onClick = {

@Composable
fun TasksPage() {
    // listas OBSERVADAS pelo Compose: adicionar, remover ou substituir um
    // elemento provoca a recomposição. (remember, e não rememberSaveable:
    // elas são perdidas se a activity for recriada, cf. exemplo 20)
    val tasks = remember {
        mutableStateListOf(Task(1, "Install Android Studio", done = true), Task(2, "Read the chapter on events"))
    }
    val log = remember { mutableStateListOf<String>() }
    var input by remember { mutableStateOf("") }
    var nextId by remember { mutableIntStateOf(3) }
    // o FocusRequester: uma forma de dar o foco ao campo (o cursor e o teclado
    // do celular) a partir do código
    val focus = remember { FocusRequester() }

    fun trace(message: String) {
        val time = LocalTime.now().format(DateTimeFormatter.ofPattern("HH:mm"))
        log.add(0, "$time $message")
    }

    // chamada pelo botão E pela tecla "OK" do teclado (keyboardActions)
    fun add() {
        val label = input.trim()
        if (label.isEmpty()) {
            return
        }
        tasks.add(Task(nextId++, label))
        trace("added \"$label\"")
        input = ""
        // o foco volta ao campo: a próxima tarefa pode ser digitada
        focus.requestFocus()
    }

    fun toggle(task: Task, done: Boolean) {
        // SUBSTITUÍMOS a tarefa por uma cópia modificada
        tasks[tasks.indexOf(task)] = task.copy(done = done)
        trace("\"${task.label}\": ${if (done) "done" else "to do"}")
    }

    fun remove(task: Task, how: String) {
        tasks.remove(task)
        trace("removed \"${task.label}\" ($how)")
    }

    // a densidade da tela: para converter pixels em dp
    val density = LocalDensity.current.density

    ExamplePage("Example 04 - events") {
        Row(verticalAlignment = Alignment.CenterVertically) {

                value = input,
                onValueChange = { input = it },

                singleLine = true,

                keyboardActions = KeyboardActions(onDone = { add() }),

data class Profile(
    val title: String,
    val firstName: String,
    val lastName: String,
    val age: Int,
    val newsletter: Boolean,
    val email: String,
    val country: String,
    val presentation: String,

    // o valor de um campo, designado por seu nome
    fun value(f: Field): Any = when (f) {
        Field.TITLE -> title
        Field.FIRSTNAME -> firstName
        Field.LASTNAME -> lastName
        Field.AGE -> age
        Field.NEWSLETTER -> newsletter
        Field.EMAIL -> email
        Field.COUNTRY -> country
        Field.PRESENTATION -> presentation
    }

    fun withValue(f: Field, v: Any): Profile = when (f) {
        Field.TITLE -> copy(title = v as String)
        Field.FIRSTNAME -> copy(firstName = v as String)
        Field.LASTNAME -> copy(lastName = v as String)
        Field.AGE -> copy(age = v as Int)
        Field.NEWSLETTER -> copy(newsletter = v as Boolean)
        Field.EMAIL -> copy(email = v as String)
        Field.COUNTRY -> copy(country = v as String)
        Field.PRESENTATION -> copy(presentation = v as String)
    }

    fun toJson(): JsonObject = JsonObject(

                is Int -> JsonPrimitive(v) // "is": um teste de tipo; v é então um Int

                else -> JsonPrimitive(v.toString())

data class State(val values: Profile, val touched: Set<Field>)

// as ações possíveis. "sealed": não há outras além das
// declaradas aqui, o compilador sabe disso
sealed interface ProfileAction {
    data class Update(val field: Field, val value: Any) : ProfileAction
    data class Leave(val field: Field) : ProfileAction
    data object Uppercase : ProfileAction // um objeto único: a ação não tem dados
    data object Reset : ProfileAction
}

// o reducer: ele NUNCA modifica o estado recebido, retorna um novo.
// Em cada ramo do when, action tem o tipo do ramo ("smart cast"):
// action.field é acessível para um Update
fun reducer(state: State, action: ProfileAction): State = when (action) {
    is ProfileAction.Update -> state.copy(values = state.values.withValue(action.field, action.value))
    is ProfileAction.Leave -> state.copy(touched = state.touched + action.field)
    ProfileAction.Uppercase -> state.copy(
        values = state.values.copy(firstName = state.values.firstName.uppercase(), lastName = state.values.lastName.uppercase()),
    )
    ProfileAction.Reset -> State(initial, emptySet())
}

        a.lastName.length > 20 -> put(Field.LASTNAME, "The last name must not exceed 20 characters.")
    }
    when {
        a.login.isEmpty() -> put(Field.LOGIN, "The login is required.")
        !loginFormat.matches(a.login) -> put(Field.LOGIN, "Unaccented lower-case letters, digits, dot, hyphen, underscore.")

import androidx.compose.runtime.getValue

// typealias: um nome mais expressivo para um tipo de função
typealias Rule = (String) -> String?

// FÁBRICAS de regras: cada uma retorna uma função de validação
fun required(message: String): Rule = { v -> if (v.isBlank()) message else null }

// uma regra que COMPLETA outra: "?:" (o operador Elvis) dá o
// resultado de other, se ele não for null, senão o do restante
fun maximum(max: Int, other: Rule): Rule = { v -> other(v) ?: if (v.length > max) "$max characters at most." else null }

class FormField(initialValue: String = "", private val rule: Rule = { null }) {
    // o estado do campo: observado pelo Compose, modificável somente AQUI (private set)
    var value by mutableStateOf(initialValue)
        private set
    var error by mutableStateOf<String?>(null)
        private set

    // o usuário já modificou o campo?
    private var modified = false

    // uma tecla digitada: o campo se valida assim que o usuário o modificou
    fun change(newValue: String) {
        value = newValue
        modified = true
        error = rule(value)
    }

    fun validate(): Boolean {
        error = rule(value)
        return error == null
    }

    // revalida, somente se o usuário já modificou o campo
    fun revalidate() {
        if (modified) validate()
    }
}

// vararg: qualquer número de campos, Form(firstName, lastName, login...)
class Form(private vararg val fields: FormField) {
    // map primeiro (TODOS os campos são validados e exibem seu erro),
    // all depois (todos estão corretos?)
    fun validate(): Boolean = fields.map { it.validate() }.all { it }
}

import kotlinx.serialization.json.JsonObject
import kotlinx.serialization.json.JsonPrimitive
import java.text.NumberFormat
import java.time.LocalDate
import java.time.LocalDateTime
import java.time.format.DateTimeFormatter
import java.time.format.FormatStyle
import java.util.Locale

// datas: formatos "prontos" (SHORT, MEDIUM, LONG, FULL), ou um padrão
val shortDate: DateTimeFormatter = DateTimeFormatter.ofLocalizedDate(FormatStyle.MEDIUM).withLocale(Locale.US)
val fullDate: DateTimeFormatter = DateTimeFormatter.ofLocalizedDate(FormatStyle.FULL).withLocale(Locale.US)
val timeOnly: DateTimeFormatter = DateTimeFormatter.ofLocalizedTime(FormatStyle.MEDIUM).withLocale(Locale.US)
val frenchDay: DateTimeFormatter = DateTimeFormatter.ofLocalizedDate(FormatStyle.LONG).withLocale(Locale.FRANCE)
val pattern: DateTimeFormatter = DateTimeFormatter.ofPattern("MM/dd/yyyy HH:mm", Locale.US)

// números e valores monetários
val euros: NumberFormat = NumberFormat.getCurrencyInstance(Locale.US).apply { currency = java.util.Currency.getInstance("EUR") }
val integer: NumberFormat = NumberFormat.getNumberInstance(Locale.US)

// apply: configura o objeto recém-criado e o retorna
val percent: NumberFormat = NumberFormat.getPercentInstance(Locale.US).apply { minimumFractionDigits = 1 }

const val DAY = "2026-10-05" // uma data no formato da API RdvMedecins
const val PRICE = 1234.5
const val RATE = 0.055
const val NAME = "marie pelissier"
val doctor = mapOf("title" to "Mme", "firstName" to "Marie", "lastName" to "PELISSIER")
val languages = listOf("Kotlin", "Compose", "Dart", "Swift", "SQL")
val slots = listOf(Slot(8, 0, 8, 20), Slot(8, 20, 8, 40), Slot(14, 5, 14, 25))

class Ex09Activity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent { ExamplesTheme { FormattingPage() } }
    }
}

@Composable
fun FormattingPage() {
    var now by remember { mutableStateOf(LocalDateTime.now()) }

    // (código exibido, valor): uma linha da tabela, como um par
    val rows = listOf(
        "shortDate.format(now)" to shortDate.format(now),
        "fullDate.format(now)" to fullDate.format(now),
        "timeOnly.format(now)" to timeOnly.format(now),
        "pattern.format(now)" to pattern.format(now),
        "fullDate.format(LocalDate.parse(DAY))" to fullDate.format(LocalDate.parse(DAY)),
        "frenchDay.format(LocalDate.parse(DAY))" to frenchDay.format(LocalDate.parse(DAY)),
        "now.toString()" to now.toString(),
        "euros.format(PRICE)" to euros.format(PRICE),
        "integer.format(PRICE)" to integer.format(PRICE),
        "\"%.1f\".format(Locale.ROOT, PRICE)" to "%.1f".format(Locale.ROOT, PRICE),
        "percent.format(RATE)" to percent.format(RATE),
        "NAME.uppercase()" to NAME.uppercase(),
        "NAME.capitalized" to NAME.capitalized,
        "\"RDV\".lowercase()" to "RDV".lowercase(),
        "languages.take(3).joinToString(\", \")" to languages.take(3).joinToString(", "),
        "JsonObject(...).toString()" to JsonObject(doctor.mapValues { JsonPrimitive(it.value) }).toString(),
        // duas transformações seguidas: o resultado da primeira vai para a segunda
        "NAME.capitalized.initials" to NAME.capitalized.initials,
    )

    ExamplePage("Example 09 - formatting") {
        // uma tabela: a primeira coluna ocupa 55 % da largura (weight)
        Row(Modifier.padding(4.dp)) {
            Text("In the code", Modifier.weight(0.55f), fontWeight = FontWeight.Bold)
            Text("Displayed", Modifier.weight(0.45f), fontWeight = FontWeight.Bold)
        }
        // um par pode ser "desestruturado": (code, value)
        for ((code, value) in rows) {
            HorizontalDivider()
            Row(Modifier.padding(4.dp)) {
                Text(code, Modifier.weight(0.55f), fontFamily = FontFamily.Monospace, fontSize = 11.sp, color = Color(0xFFC7254E))
                Text(value, Modifier.weight(0.45f))
            }
        }
        Spacer(Modifier.height(16.dp))
        Text("Going through the entries of a Map", style = MaterialTheme.typography.titleSmall)
        for ((key, value) in doctor) {
            Text("$key = $value")
        }
        Spacer(Modifier.height(16.dp))
        Text("A home-made function: timeRange", style = MaterialTheme.typography.titleSmall)
        for (s in slots) {
            Text("${timeRange(s)} (start: ${timeRange(s, Part.START)})")
        }
        Spacer(Modifier.height(16.dp))
        OutlinedButton(onClick = { now = LocalDateTime.now() }) { Text("Refresh the time") }
    }
}
    • linhas 51–62: os formatadores, criados de uma vez por todas fora das funções [@Composable] (criá-los custa mais do que usá-los). [FormatStyle.MEDIUM], [FULL]...: formatos “prontos”; [ofPattern]: um padrão; [apply [ ... ]] configura o objeto recém-criado e o retorna;
    • linhas 85–103: cada linha da tabela é um par (código exibido, valor), gerado por [to];
    • linha 113: um par, como uma entrada de [Map], é “desempacotado”: [for ((code, value) in rows)].

    Observe ["%.1f".format(Locale.ROOT, PRICE)]: sem o idioma “neutro” [Locale.ROOT], [format] usaria o idioma do celular e escreveria “1234,5” em francês. É o que se deseja para exibição, não para dados enviados a um servidor.
